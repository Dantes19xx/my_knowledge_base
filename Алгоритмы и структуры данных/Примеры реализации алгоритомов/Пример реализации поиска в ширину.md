BFS, Breadth-First Search

```python
from collections import deque

def bfs(graph, start):
    visited = set()            # множество посещённых вершин
    queue = deque([start])     # очередь для BFS

    while queue:
        vertex = queue.popleft()  # извлекаем вершину из очереди
        if vertex not in visited:
            print(vertex)         # посещаем вершину (можно заменить на другую обработку)
            visited.add(vertex)   # помечаем как посещённую
            queue.extend(graph[vertex] - visited)  # добавляем соседей в очередь

# Пример графа (ориентированный или неориентированный)
graph = {
    'A': {'B', 'C'},
    'B': {'A', 'D', 'E'},
    'C': {'A', 'F'},
    'D': {'B'},
    'E': {'B', 'F'},
    'F': {'C', 'E'}
}

bfs(graph, 'A')
```
